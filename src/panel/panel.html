  1. 1. Enable per-origin globalThis.Signal injection

export default defineConfig({ plugins: [signalsDevtoolsPlugin()] });
  3. 3. Register a private runtime or Watcher directly 
    globalThis.__SIGNALS_DEVTOOLS_HOOK__?.register({
  name: "app",
  Signal,
  watchers: [watcher],
